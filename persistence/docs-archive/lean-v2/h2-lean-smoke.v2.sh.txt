#!/usr/bin/env bash
# 精简版（Lean v2）V1 DDL 冒烟：在 H2 1.4.200 (MODE=MySQL) 上执行 V1__lean_baseline.sql、一个正例（含断言）、约束负例与对照组。
# 用法：bash persistence/ops/h2-lean-smoke.sh   （需 JDK 17+ 与本地仓库中的 h2-1.4.200.jar；可用 H2_JAR 覆盖）
# 只证明 H2 语法与 CHECK/UNIQUE/FK 行为；不证明 MySQL 的排序规则、长度、锁与事务语义 [未验证-MySQL]。
#
# v2 变化（评审 G3）：
#   * 负例必须以"预期的 H2 错误码"失败才算通过（23513 CHECK、23505 唯一、23506 外键缺父、23503 外键有子、
#     22001 超长、22003 数值越界）；以其他原因失败（如列数不符、语法错误）记为 FAIL(wrong error)。
#   * 正例断言改为向带 CHECK(ok=1) 的 smoke_assert 表插入结果，MySQL 8.0.16+ 同样会报错（不再依赖 H2 的除零报错）。
#   * MySQL 上复跑时预期错误码对应：23513→3819、23505→1062、23506→1452、23503→1451、22001→1406、22003|22004→1264
#     （1406/1264 仅在 STRICT_TRANS_TABLES 下报错，非严格模式会截断——见设计 §7.3）。
set -u
HERE="$(cd "$(dirname "$0")" && pwd)"
JAR="${H2_JAR:-$HOME/.m2/repository/com/h2database/h2/1.4.200/h2-1.4.200.jar}"
URL="jdbc:h2:mem:t;MODE=MySQL;DATABASE_TO_LOWER=TRUE;CASE_INSENSITIVE_IDENTIFIERS=TRUE"
TMP="$(mktemp -d)"
sed 's/\${table_options}//' "$HERE/../src/main/resources/db/migration/V1__lean_baseline.sql" > "$TMP/v1.sql"
cat "$TMP/v1.sql" "$HERE/h2-lean-smoke-positive.sql" > "$TMP/t.sql"
run() { java -cp "$JAR" org.h2.tools.RunScript -url "$URL" -script "$1" >"$TMP/out.txt" 2>&1; }
if run "$TMP/t.sql"; then echo "POSITIVE OK"; else echo "POSITIVE FAILED"; cat "$TMP/out.txt"; rm -rf "$TMP"; exit 1; fi
fails=0; total=0
# neg <名称> <预期错误码正则> <SQL>
neg() {
  total=$((total+1)); cp "$TMP/t.sql" "$TMP/n.sql"; echo "$3" >> "$TMP/n.sql"
  if run "$TMP/n.sql"; then echo "FAIL(accepted): $1"; fails=$((fails+1));
  elif grep -Eq "\[($2)-200\]" "$TMP/out.txt"; then echo "ok rejected [$2]: $1";
  else echo "FAIL(wrong error, want $2): $1"; grep -Eo "\[[0-9A-Z]+-200\]" "$TMP/out.txt" | head -1; fails=$((fails+1)); fi
}
# ctl <名称> <SQL>：与负例只差一处的合法语句必须被接受（防止负例因语法错误而"假通过"）
ctl() {
  total=$((total+1)); cp "$TMP/t.sql" "$TMP/c.sql"; echo "$2" >> "$TMP/c.sql"
  if run "$TMP/c.sql"; then echo "ok accepted: $1"; else echo "FAIL(rejected control): $1"; head -2 "$TMP/out.txt"; fails=$((fails+1)); fi
}
