-- =====================================================================
-- millionnaire 持久化层 V1 基线（精简版 · 测试版 v4）
-- 设计说明：.discuss/db-design-lean.md（v4 测试版）；收口说明：.discuss/db-lean-trim-note.md
-- 精简版 v2（含 service_control、boot_id 等，已收口删除）归档在 persistence/docs-archive/lean-v2/
-- 完整版（18 表，已废弃）归档在 persistence/docs-archive/V1__full_baseline.sql.txt
--
-- 前提：对局状态只在内存；测试阶段不做发版流程、重启善后与故障恢复。重启即重启，
--       上一个进程留下的 OPEN 房间不清扫（孤儿行），外壳按号加入时以内存为准。
--       本库只保存：账号与微信身份、房间号与房间元数据、局结束时的战绩、（可选）对局公开事件日志、（可选）聊天。
--
-- 方言约定：
--   * 以MySQL5.7/8.0、InnoDB为准；5.7忽略CHECK，Store须验证全部谓词（设计§1.1）；同时限定在 H2 (MODE=MySQL) 可执行的子集。
--     H2 1.4.200 已执行通过（persistence/ops/h2-lean-smoke.sh）；真实 MySQL 未验证 [未验证-MySQL5.7]。
--   * 表选项经 Flyway 占位符 ${table_options} 注入（Maven 资源过滤必须对 db/migration 关闭）：
--       MySQL: ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_bin
--       H2   : 空串
--     utf8mb4_bin为PAD SPACE，精确身份不依赖文本排序。
--     需要"按字节精确相等"的标识符（openid、appid、建房请求 ID）一律 VARBINARY，不依赖排序规则。
--   * 二进制长度检查用 LENGTH()（MySQL = 字节数）；H2 1.4.200 的 OCTET_LENGTH(VARBINARY) 返回 2 倍字节数，故不用。
--   * 时间一律 UTC epoch 毫秒 BIGINT（*_at），由应用时钟写入。
--   * 不用JSON/窗口函数/CTE/DEFAULT表达式；CHECK与宽度规则须应用校验/单测，5.7反转清单见ops/mysql57-check-reversal.md。
--   * 最大显式索引键96字节，低于5.7的767/3072；最近索引全升序、查询反向扫描。
--   * OPEN幂等仅复用本进程对象/仍持有原上下文，否则报解散，绝不据DB创世。
--   * 当前文字聊天只存内存，本表为可选契约，当前不接线。
--   * 索引单独 CREATE INDEX；外键默认 RESTRICT，无级联。外键共 5 个（见各表）。
--   * 可空列的 CHECK 一律写显式 IS [NOT] NULL 分支（CHECK 对 UNKNOWN 放行）。
--   * MySQL DDL 隐式提交，本文件不是可回滚单元：基线中途失败时库内只有本文件的对象 →
--     按设计 §1"迁移"行的人工受控流程（核验库名、确认未开放服务、确认无业务数据）删库重建后重跑；绝不在启动失败时自动 DROP。
-- =====================================================================

-- ---------------------------------------------------------------------
-- 1. 用户与微信身份
-- ---------------------------------------------------------------------
CREATE TABLE app_user (
  user_id        BIGINT        NOT NULL,              -- 随机正整数（SecureRandom，<= 2^53-1，JS 安全）；引擎 playerId = 十进制字符串；墓碑永久保留 → 永不复用
  nickname       VARCHAR(32)   NOT NULL,              -- 可重复；trim后1..32 UTF-16码元与完整引擎规则由应用校验（设计§2.1）；注销后改为固定占位
  avatar_id      SMALLINT      NOT NULL DEFAULT 0,
  status         VARCHAR(8)    NOT NULL DEFAULT 'ACTIVE',
  created_at     BIGINT        NOT NULL,              -- 注销时改写为 deleted_at（墓碑不保留注册/登录时间）
  updated_at     BIGINT        NOT NULL,
  last_login_at  BIGINT        NOT NULL,              -- 同上
  deleted_at     BIGINT        NULL,
  live_mark      TINYINT       NOT NULL DEFAULT 1,    -- 1 = 未注销，0 = 已注销。仅为被个人数据外键引用（见 game_record_player / chat_message）
  CONSTRAINT pk_app_user PRIMARY KEY (user_id),
  CONSTRAINT uk_app_user_live UNIQUE (user_id, live_mark),   -- 外键引用目标：子行 (user_id, 1) 只能指向未注销用户
  CONSTRAINT ck_app_user_id CHECK (user_id > 0 AND user_id <= 9007199254740991),
  CONSTRAINT ck_app_user_avatar CHECK (avatar_id >= 0 AND avatar_id <= 999),
  CONSTRAINT ck_app_user_status CHECK ((status IN ('ACTIVE', 'BANNED') AND deleted_at IS NULL AND live_mark = 1)
                                    OR (status = 'DELETED' AND deleted_at IS NOT NULL AND live_mark = 0))
) ${table_options};

-- openid -> 用户。注销时删除本行（再登录即新用户）。不存 session_key，不存 unionid（最小必要）。
CREATE TABLE wx_identity (
  app_id         VARBINARY(32) NOT NULL,              -- ASCII 字节，精确相等
  open_id        VARBINARY(64) NOT NULL,
  user_id        BIGINT        NOT NULL,
  created_at     BIGINT        NOT NULL,
  last_login_at  BIGINT        NOT NULL,
  CONSTRAINT pk_wx_identity PRIMARY KEY (app_id, open_id),
  CONSTRAINT uk_wx_identity_user UNIQUE (user_id, app_id),
  CONSTRAINT fk_wx_identity_user FOREIGN KEY (user_id) REFERENCES app_user (user_id),            -- 外键 1
  CONSTRAINT ck_wx_identity_len CHECK (LENGTH(app_id) >= 1 AND LENGTH(open_id) >= 1)
) ${table_options};

-- ---------------------------------------------------------------------
-- 2. 房间：房间号唯一 + 关闭后失效 + 冷却（无单独的号码表）
--    held_code：房间 OPEN 时等于 room_code；关闭后仍占着号码直到被新建房事务在冷却期满后释放（置 NULL）。
--    UNIQUE(held_code) 保证"占用中或冷却中"的号码全局唯一；多个 NULL 不冲突（MySQL 与 H2 均如此）。
--    按号查房只认 status='OPEN'，且外壳必须再核验内存房间存在且可加入（重启后上一进程的 OPEN 行是孤儿，见设计 §3.3）。
--    测试版不记录建房进程（v2 的 boot_id 已删除，启动清扫在后补清单中）。
-- ---------------------------------------------------------------------
CREATE TABLE room (
  room_id           BIGINT        NOT NULL,           -- 随机正整数；引擎 roomId = 十进制字符串；建房时还须核验room/game_record/game_log/chat_message（含保全）及本进程待写引用无同号（设计 §1）
  room_code         INT           NOT NULL,           -- 六位房间号（展示时补零）；历史值
  held_code         INT           NULL,               -- 见上
  created_by        BIGINT        NOT NULL,
  create_request_id VARBINARY(16) NOT NULL,           -- 建房幂等键（客户端 UUID 的 16 字节）
  status            VARCHAR(8)    NOT NULL,           -- OPEN / CLOSED（大厅/对局中的区别只在内存）
  max_saved_game_no INT           NOT NULL DEFAULT 0, -- 已保存战绩中的最大局号（仅供排查；不是局数，也不是开局编号来源）
  created_at        BIGINT        NOT NULL,
  updated_at        BIGINT        NOT NULL,
  closed_at         BIGINT        NULL,
  close_reason      VARCHAR(16)   NULL,
  CONSTRAINT pk_room PRIMARY KEY (room_id),
  CONSTRAINT uk_room_held_code UNIQUE (held_code),
  CONSTRAINT uk_room_create_request UNIQUE (created_by, create_request_id),   -- 左前缀兼作外键索引与注销定位
  CONSTRAINT fk_room_creator FOREIGN KEY (created_by) REFERENCES app_user (user_id),            -- 外键 2
  CONSTRAINT ck_room_id CHECK (room_id > 0 AND room_id <= 9007199254740991),
  CONSTRAINT ck_room_code CHECK (room_code >= 0 AND room_code <= 999999),
  CONSTRAINT ck_room_held CHECK (held_code IS NULL OR held_code = room_code),
  CONSTRAINT ck_room_ids CHECK (LENGTH(create_request_id) = 16 AND max_saved_game_no >= 0),
  -- 关闭原因：测试版实际只写 ENGINE（引擎 RoomClosed）、IDLE（7 天无人）、FAULT（内核故障停房）、ADMIN；
  -- UPGRADE / RESTART 保留为合法取值（后补的发版与启动清扫会用到），测试版不依赖。CLOSED 时原因必须非空。
  CONSTRAINT ck_room_status CHECK (
       (status = 'OPEN' AND held_code IS NOT NULL AND closed_at IS NULL AND close_reason IS NULL)
    OR (status = 'CLOSED' AND closed_at IS NOT NULL AND close_reason IS NOT NULL
        AND close_reason IN ('ENGINE', 'IDLE', 'UPGRADE', 'RESTART', 'FAULT', 'ADMIN')))
) ${table_options};

CREATE INDEX ix_room_closed ON room (closed_at);              -- 关闭房间的保留期清理

-- ---------------------------------------------------------------------
-- 3. 战绩：局结束（正常结束或中止）时一次性写入 game_record + 允许保留的 game_record_player，同一短事务。
--    幂等键：UNIQUE(room_id, game_no)；重复时比较 draft_sha256。表头至少保留 30 天（幂等保留边界，设计 §4.2）。
--    跨表一致性（人数/座位/结算与表头一致、时间一致）不由 DDL 保证，由 RecordStore 写前校验草稿（设计 §3.5）。
-- ---------------------------------------------------------------------
CREATE TABLE game_record (
  record_id         BIGINT        NOT NULL AUTO_INCREMENT,
  room_id           BIGINT        NOT NULL,           -- 不设外键：房间行到期清理后战绩仍保留
  game_no           INT           NOT NULL,           -- 引擎 gameNo（long）；超出 INT 的草稿被拒绝，不截断
  outcome           VARCHAR(8)    NOT NULL,           -- FINISHED（GameEnded）/ ABORTED（GameAborted；测试版外壳不发 EndGame，故实际不产生，保留取值）
  end_reason        VARCHAR(32)   NOT NULL,           -- FINISHED：引擎 reason（当前 TIME_UP / NO_PLAYERS / LAST_SURVIVOR / ALL_ELIMINATED / ALL_AWAY）；ABORTED：EndGame.reason
  end_mode          VARCHAR(16)   NOT NULL,           -- 引擎 EndMode 名
  time_limit_min    SMALLINT      NULL,               -- 限时模式的分钟数；破产模式为 NULL
  board_id          VARCHAR(32)   NOT NULL,
  initial_cash      BIGINT        NOT NULL,
  player_count      SMALLINT      NOT NULL,           -- 开局座位数；不因注销/修剪减少
  started_at        BIGINT        NOT NULL,           -- GameStarted.startedAt
  ended_at          BIGINT        NOT NULL,           -- 终局事件的业务时间（产生它的 TaskFired.at 或 InputAccepted.at）
  engine_version    VARCHAR(64)   NOT NULL,
  config_hash       VARCHAR(128)  NOT NULL,           -- 引擎 configHash() 文本
  draft_sha256      VARBINARY(32) NOT NULL,           -- 规范草稿（含全部座位，注销跳过之前）的 SHA-256；幂等命中时比较
  created_at        BIGINT        NOT NULL,
  CONSTRAINT pk_game_record PRIMARY KEY (record_id),
  CONSTRAINT uk_game_record_game UNIQUE (room_id, game_no),
  CONSTRAINT ck_game_record_outcome CHECK (outcome IN ('FINISHED', 'ABORTED')),
  CONSTRAINT ck_game_record_mode CHECK ((end_mode = 'TIME_LIMIT' AND time_limit_min IS NOT NULL AND time_limit_min > 0)
                                     OR (end_mode = 'BANKRUPTCY' AND time_limit_min IS NULL)),
  CONSTRAINT ck_game_record_values CHECK (game_no >= 1 AND player_count >= 1 AND player_count <= 8
                                          AND initial_cash >= 0 AND ended_at >= started_at
                                          AND LENGTH(draft_sha256) = 32)
) ${table_options};

CREATE INDEX ix_game_record_ended ON game_record (ended_at);   -- 孤儿表头清理（ended_at 早于 30 天且无明细）

-- 每局每人一行；玩家只看自己的行（+ 表头）。不存昵称快照（界面不展示他人，最小必要）。
-- (user_id, user_live) 外键指向 app_user(user_id, live_mark)：子行恒user_live=1且父元组合法时拒绝墓碑明细；5.7错用0须Store拒绝；
-- 注销事务必须先删本人明细再把 live_mark 置 0（否则 RESTRICT 拒绝）。这是兜底；正常路径由 RecordStore 先 FOR UPDATE 锁用户行并跳过已注销者（设计 §5.1）。
CREATE TABLE game_record_player (
  record_id         BIGINT        NOT NULL,
  seat_no           SMALLINT      NOT NULL,           -- GameStarted.seats 中的下标
  user_id           BIGINT        NOT NULL,
  ended_at          BIGINT        NOT NULL,           -- 冗余表头（相等由 RecordStore 保证），供"我的最近 20 局"走索引
  finish_rank       SMALLINT      NULL,               -- 并列 1、1、3；中止局为 NULL
  final_net_worth   BIGINT        NULL,               -- Standing.netWorth；中止局为 NULL
  final_cash        BIGINT        NULL,               -- Standing.cash；中止局为 NULL
  life_state        VARCHAR(16)   NOT NULL,           -- 终局时 ALIVE / BANKRUPT / SURRENDERED（= O21 所说"淘汰原因"的记录粒度）
  user_live         TINYINT       NOT NULL DEFAULT 1, -- 恒为 1；仅供外键
  CONSTRAINT pk_game_record_player PRIMARY KEY (record_id, seat_no),
  CONSTRAINT uk_grp_user UNIQUE (record_id, user_id),
  CONSTRAINT fk_grp_record FOREIGN KEY (record_id) REFERENCES game_record (record_id),          -- 外键 3
  CONSTRAINT fk_grp_user_live FOREIGN KEY (user_id, user_live) REFERENCES app_user (user_id, live_mark),  -- 外键 4
  CONSTRAINT ck_grp_seat CHECK (seat_no >= 0 AND seat_no < 8),
  CONSTRAINT ck_grp_live CHECK (user_live = 1),
  CONSTRAINT ck_grp_life CHECK (life_state IN ('ALIVE', 'BANKRUPT', 'SURRENDERED')),
  CONSTRAINT ck_grp_result CHECK ((finish_rank IS NULL AND final_net_worth IS NULL AND final_cash IS NULL)
                               OR (finish_rank IS NOT NULL AND finish_rank >= 1 AND finish_rank <= 8
                                   AND final_net_worth IS NOT NULL AND final_cash IS NOT NULL))
) ${table_options};

-- 最近 20 局查询：WHERE user_id=? AND user_live=1 ORDER BY ended_at DESC, record_id DESC LIMIT 20；同时满足外键 4 的子表索引要求
CREATE INDEX ix_grp_user_ended ON game_record_player (user_id, user_live, ended_at, record_id);

-- ---------------------------------------------------------------------
-- 4.（可选）对局公开事件日志：局结束时写一行，只含该局 GameStarted..终局之间的 PUBLIC 事件（无手牌、无随机状态）。
--    用于申诉/排查；保留 90 天；不加密（不含秘密，但仍是含 user_id 的用户行为记录，需最小访问权限）；不设外键。
-- ---------------------------------------------------------------------
CREATE TABLE game_log (
  room_id           BIGINT        NOT NULL,
  game_no           INT           NOT NULL,
  outcome           VARCHAR(8)    NOT NULL,
  content_kind      VARCHAR(16)   NOT NULL,           -- PUBLIC_EVENTS（v1 只允许这一种）
  codec             VARCHAR(16)   NOT NULL,           -- GZIP_EVLOG1：gzip(引擎 Codec 对 EventLog 信封的规范文本)
  engine_version    VARCHAR(64)   NOT NULL,
  config_hash       VARCHAR(128)  NOT NULL,
  event_count       INT           NOT NULL,
  plain_len         INT           NOT NULL,           -- 解压后字节数（声明上限；采集缓冲另有上限）
  plain_sha256      VARBINARY(32) NOT NULL,
  payload           MEDIUMBLOB    NOT NULL,
  ended_at          BIGINT        NOT NULL,
  created_at        BIGINT        NOT NULL,
  CONSTRAINT pk_game_log PRIMARY KEY (room_id, game_no),
  CONSTRAINT ck_game_log_kind CHECK (content_kind = 'PUBLIC_EVENTS' AND codec = 'GZIP_EVLOG1'
                                     AND outcome IN ('FINISHED', 'ABORTED')),
  CONSTRAINT ck_game_log_len CHECK (event_count >= 0 AND plain_len >= 0 AND plain_len <= 8388608
                                    AND LENGTH(plain_sha256) = 32 AND game_no >= 1)
) ${table_options};

CREATE INDEX ix_game_log_ended ON game_log (ended_at);   -- 90 天清理

-- ---------------------------------------------------------------------
-- 5.（可选）文字聊天：保留 30 天（待合规核实）；被举报消息 hold_until 保全；固定表情不入库。
--    主键 (room_id, msg_no)：msg_no 由房间串行执行器在转发时分配并随消息广播 → 客户端举报时引用的标识可信且在入库前已知。
--    (sender_user_id, sender_live) 外键：Store锁合法父元组、新插入恒sender_live=1；5.7错误0/NULL须Store拒绝；注销时保全中的消息把 sender_live 置 NULL（脱离外键，保留证据）。
-- ---------------------------------------------------------------------
CREATE TABLE chat_message (
  room_id         BIGINT        NOT NULL,
  msg_no          INT           NOT NULL,     -- 房间内从 1 递增
  sender_user_id  BIGINT        NOT NULL,
  content         VARCHAR(200)  NOT NULL,
  sec_status      VARCHAR(16)   NOT NULL,     -- 内容安全检测结果
  hold_until      BIGINT        NULL,         -- 举报保全到期（服务端决定，只延长不缩短）；非空且未到期时普通清理跳过
  sender_live     TINYINT       NULL DEFAULT 1,
  created_at      BIGINT        NOT NULL,
  CONSTRAINT pk_chat_message PRIMARY KEY (room_id, msg_no),
  CONSTRAINT fk_chat_sender_live FOREIGN KEY (sender_user_id, sender_live) REFERENCES app_user (user_id, live_mark),  -- 外键 5
  CONSTRAINT ck_chat_message CHECK (sec_status IN ('PASS', 'REVIEW', 'RISKY', 'UNCHECKED') AND msg_no >= 1),
  CONSTRAINT ck_chat_live CHECK ((sender_live IS NOT NULL AND sender_live = 1)
                              OR (sender_live IS NULL AND hold_until IS NOT NULL))
) ${table_options};

CREATE INDEX ix_chat_time ON chat_message (created_at);                                -- 30 天清理
CREATE INDEX ix_chat_sender ON chat_message (sender_user_id, sender_live, created_at); -- 注销定位；满足外键 5 的子表索引要求
